```
# env
mkdir -p ~/oisf/ && cd ~/oisf/
```
```
# ndpi
wget https://github.com/ntop/nDPI/archive/refs/tags/6.0.zip
wget https://github.com/ntop/nDPI/archive/refs/tags/4.14.tar.gz
unzip ./6.0.zip -d ./ndpi6
tar xzvf ./4.14.tar.gz
cd ./nDPI-4.14
./autogen.sh
./configure --prefix=/opt/oisf/ndpi4 --with-only-libndpi
make -j$(nproc)
sudo make install
cd ../
```
```
# suricata
wget https://www.openinfosecfoundation.org/download/suricata-8.0.8.tar.gz
tar xzvf suricata-8.0.8.tar.gz
cd ./suricata-8.0.8
```
